    async def scan_as_hex(self):
        byte_cmd = self.create_command(self.cmd_scan)
        await self.send_command(byte_cmd)
        # Read scan
        rawdata = []
        for i in range(3):
            full_resp = await self.read_response_as_raw()
            rawdata.append(full_resp)
        # Parse response (Store as base64 strings)
        hex_data = {'sample':          rawdata[0].hex(),
                    'sample_dark':     rawdata[1].hex(),
                    'sample_gradient': rawdata[2].hex()}
        print(hex_data)
        # Now interpret
        import numpy as np
        sample = np.array(struct.unpack('<bbH4x400I196x',rawdata[0]))
        sampleDark = np.array(struct.unpack('<bbH4x400I196x',rawdata[1]))
        sampleGradient = np.array(struct.unpack('<bbH4x400I52x',rawdata[2]))
        print('sample', sample)
        print('sampleDark', sampleDark)
        print('sampleGradient', sampleGradient)
        correctedSample = sample - sampleDark
        correctedGradient = sampleGradient - sampleDark
        reflectance = correctedSample / correctedGradient
        print('Reflectance', reflectance)
        return(hex_data)
